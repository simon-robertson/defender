import {builder,sceneryModels} from './first-level.js';
import {canonicalModel} from '../shared/model.js';
import {newLevel,validateLevel} from '../shared/level.js';
import {TerrainField} from '../shared/terrain.js';
import {defaultWaterMaterial} from '../shared/water.js';

const palettes={
 B:['#77523B','#A4774C','#254D4A','#3F6D5D','#6F906D','#9DB28A','#CB8C52','#A96743','#E5AD70','#C4C28A','#AD5842'],
 C:['#595347','#7B715D','#263D34','#3A5748','#D6DED8','#F1F2EA','#85939B','#5D6D77','#B7C3C5','#E8E2D2','#A4815D'],
 D:['#4C4141','#726255','#3F3047','#654660','#875B79','#A47491','#85858C','#5C5C67','#AFACA9','#B9C8B8','#A47C69'],
};
const prefixes={B:'ochre',C:'ice',D:'plum'};
const materials=zone=>palettes[zone].map((color,i)=>({name:['Trunk','Trunk light','Plant deep','Plant','Plant cap','Plant highlight','Stone','Stone shade','Stone light','Flower','Accent'][i],color,emission:0,reflectiveness:0}));

export function zoneSceneryModels(){
 const originals=sceneryModels(),result=[];
 for(const zone of ['B','C','D']){
  const prefix=prefixes[zone],palette=materials(zone);
  const copy=(source,suffix,name)=>{
   const model=structuredClone(originals.find(m=>m.id===source));model.id=prefix+'-'+suffix;model.name=name;model.materials=structuredClone(palette);
   result.push(canonicalModel(model));
  };
  copy('scenery-ruin-wall','wall',zone==='B'?'Sandstone · broken wall':zone==='C'?'Snow-capped · blue stone wall':'Violet stone · broken wall');
  copy('scenery-rock','rock',zone==='B'?'Sandstone · layered outcrop':zone==='C'?'Snow-capped · slate outcrop':'Plum stone · outcrop');
  copy('detail-grass','tuft',zone==='B'?'Desert · dry scrub':zone==='C'?'Frost · low sprigs':'Plum · low shoots');
  copy('detail-pebbles','pebbles',zone==='B'?'Sandstone · fragments':zone==='C'?'Blue stone · fragments':'Violet stone · fragments');
  copy('detail-flowers','flowers',zone==='B'?'Desert · small flowers':zone==='C'?'Frost · pale flowers':'Mint · small flowers');
  copy('scenery-shrub','shrub',zone==='B'?'Desert · low shrub':zone==='C'?'Snow-capped · dwarf fir':'Plum · low shrub');
  if(zone==='B'){
   const b=builder('ochre-cactus','Desert · branching cactus',45,palette);
   b.crown(9,0,9,15,40,15,3,1);b.box(10,38,10,14,44,14,4);b.box(10,2,9,11,38,10,4);
   b.box(4,16,10,10,21,14,3);b.crown(3,17,9,7,32,15,3,1);b.box(4,30,10,6,34,14,4);
   b.box(14,23,10,20,28,14,3);b.crown(18,24,9,22,37,15,3,1);b.box(19,35,10,21,39,14,4);result.push(b.finish());
   const c=builder('ochre-mesa','Sandstone · stepped mesa',46,palette);
   c.crown(1,0,2,23,13,23,7,3);c.crown(2,12,3,22,28,22,6,3);c.crown(4,27,4,21,40,21,6,3);c.crown(5,39,5,20,45,20,8,2);
   c.box(6,14,3,13,16,4,8);c.box(4,29,8,5,32,18,7);result.push(c.finish());
  }else if(zone==='C'){
   const b=builder('ice-fir','Snow-capped · tiered fir',56,palette);
   b.box(10,0,10,14,46,14,0);
   for(const [y,r,h]of [[10,10,10],[24,8,9],[37,5,8],[46,2,7]]){
    b.crown(12-r,y,12-r,12+r,y+h,12+r,3,Math.max(1,Math.floor(r/3)));
    b.crown(13-r,y+h-2,13-r,11+r,y+h+1,11+r,5,Math.max(0,Math.floor(r/3)-1));
   }result.push(b.finish());
   const c=builder('ice-pillar','Snow-capped · slate pillar',38,palette);
   c.crown(3,0,3,21,11,21,7,3);c.crown(5,10,5,19,29,19,6,2);c.crown(4,28,4,20,33,20,8,2);c.crown(5,33,5,19,37,19,5,2);result.push(c.finish());
  }else{
   copy('scenery-oak','tree','Plum · broad canopy');copy('scenery-oak-tall','tree-tall','Plum · tall canopy');
   const b=builder('plum-arch','Violet ruins · arch fragment',40,palette);
   b.box(2,0,8,8,4,17,7);b.box(16,0,8,22,4,17,7);b.box(3,4,9,7,31,16,6);b.box(17,4,9,21,31,16,6);
   b.box(2,13,8,8,16,17,8);b.box(16,13,8,22,16,17,8);b.box(2,28,8,22,33,17,8);b.box(4,33,9,20,37,16,6);b.box(6,37,10,15,39,15,4);result.push(b.finish());
  }
 }
 return result;
}

export function zonePreviewLevel(zone){
 const prefix=prefixes[zone];if(!prefix)throw new Error('Use the existing first-level source for zone A.');
 const l=newLevel(prefix+'-01');l.zone=zone;l.size=[14,10];
 l.name={B:'The sunken aqueduct',C:'The frostwater pass',D:'The violet cloister'}[zone];
 const corners={B:[[0,3],[4,3],[4,6],[9,6],[9,2],[13,2]],C:[[0,5],[3,5],[3,2],[8,2],[8,6],[13,6]],D:[[0,2],[4,2],[4,6],[9,6],[9,3],[13,3]]}[zone];
 let [x,z]=corners[0];l.path.push([x,0,z]);for(const [endX,endZ]of corners.slice(1))while(x!==endX||z!==endZ){x+=Math.sign(endX-x);z+=Math.sign(endZ-z);l.path.push([x,0,z]);}
 l.environment={timeOfDay:10.5,weather:{enabled:false},sky:{direction:[-.65,1,-.45],color:{B:'#FFE4BC',C:'#FFF8EB',D:'#FFF8EB'}[zone],intensity:.88,ambient:{B:'#D4BBA0',C:'#C7CFD0',D:'#CBCBC5'}[zone],ambientIntensity:.48}};
 l.waterMaterial={...defaultWaterMaterial(),color:{B:'#408B91',C:'#52899F',D:'#5E839B'}[zone]};
 const terrain=new TerrainField(l);for(let bx=0;bx<14;bx++)terrain.setBox(bx,8,4,12);
 for(let bx=0;bx<14;bx++)terrain.setBox(bx,0,zone==='B'?32+(bx%3)*8:zone==='C'?32+(bx%4)*4:28+(bx%3)*4,0);
 for(const [bx,bz]of [[0,6],[0,7],[1,7],[12,7],[13,7]])if(!l.path.some(p=>p[0]===bx&&p[2]===bz))terrain.setBox(bx,bz,32,0);
 terrain.commit();
 const used=new Set(l.path.map(p=>p[0]+','+p[2]));let id=0;
 const place=(model,x,z,turn=0,variant='base',offset=null)=>{
  const key=x+','+z;if(used.has(key)||z===8)return false;if(!offset)used.add(key);
  l.placements.push({id:'scene-'+(++id),model,variant,box:[x,0,z],turn,...(offset?{kind:'detail',offset}: {})});return true;
 };
 const towers={B:[[2,5],[7,4],[11,4]],C:[[2,3],[6,4],[11,4]],D:[[2,4],[7,4],[11,5]]}[zone];
 towers.forEach(([x,z],i)=>place('turret-study',x,z,0,i===1?'upgrade1':'base'));
 const tall=zone==='B'?'mesa':zone==='C'?'fir':'tree-tall',small=zone==='B'?'cactus':zone==='C'?'fir':'tree';
 for(let x=0;x<14;x++)place(prefix+'-'+(x%3===0?small:tall),x,0,x%4);
 for(const [x,z]of [[0,1],[1,1],[5,1],[7,1],[11,1],[13,1],[0,4],[0,6],[0,7],[1,7],[12,7],[13,7],[2,9],[4,9],[8,9],[11,9]])place(prefix+'-'+small,x,z,(x+z)%4);
 for(const [x,z,turn]of [[1,2,0],[2,2,0],[5,2,0],[6,2,0],[10,1,1],[11,2,1],[12,5,1],[1,6,0],[6,7,0],[7,7,0],[10,7,0]])place(prefix+'-wall',x,z,turn);
 for(const [x,z]of [[2,1],[9,1],[1,4],[10,5],[11,7]])place(prefix+'-'+(zone==='B'?'mesa':zone==='C'?'pillar':'arch'),x,z,(x+z)%4);
 for(const [x,z]of [[3,1],[6,1],[8,1],[12,1],[2,7],[4,7],[9,7],[0,9],[6,9],[13,9]])place(prefix+'-rock',x,z,(x+z)%4);
 for(let z=1;z<8;z++)for(let x=0;x<14;x++)if((x*7+z*11)%5===0)place(prefix+'-shrub',x,z,(x+z)%4);
 for(let z=1;z<10;z++)for(let x=0;x<14;x++){
  if(used.has(x+','+z)||z===8)continue;const n=(x*37+z*19)%9;
  place(prefix+'-'+(n<3?'flowers':n<6?'tuft':'pebbles'),x,z,0,'base',[-5,0,-4]);
  if(n%2===0)place(prefix+'-tuft',x,z,0,'base',[5,0,5]);
 }
 return l;
}
if(import.meta.main){
 const models=zoneSceneryModels(),library=new Map(models.map(m=>[m.id,m]));library.set('turret-study',await Bun.file('assets/models/turret-study.json').json());
 for(const zone of ['B','C','D'])validateLevel(zonePreviewLevel(zone),library);
 for(const model of models){const path=`assets/models/${model.id}.json`;if(!await Bun.file(path).exists())await Bun.write(path,JSON.stringify(model,null,2)+'\n');}
 for(const zone of ['B','C','D']){const level=zonePreviewLevel(zone),path=`assets/levels/${level.id}.json`;if(!await Bun.file(path).exists())await Bun.write(path,JSON.stringify(level,null,2)+'\n');console.log(`${zone}: ${level.name}, ${level.path.length} path boxes, ${level.placements.length} placements`);}
 console.log(`${models.length} zone scenery sources; existing files preserved.`);
}
